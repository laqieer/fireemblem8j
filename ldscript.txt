OUTPUT_ARCH(arm)

MEMORY
{
    rom   : ORIGIN = 0x08000000, LENGTH = 32M
    iwram : ORIGIN = 0x03000000, LENGTH = 32K
    ewram : ORIGIN = 0x02000000, LENGTH = 256K
}

__text_start  = ORIGIN(rom);
__rom_start   = ORIGIN(rom);
__ewram_start = ORIGIN(ewram);
__iwram_start = ORIGIN(iwram);

ENTRY(Init)

/* crt0 immediately follows the 0xC0-byte ROM header; defined here so the
   header's `b crt0` resolves while crt0 itself is still inside the baseline. */
crt0 = 0x080000C0;

SECTIONS
{
    .rom __rom_start : ALIGN(4)
    {
        /* Decompiled objects are spliced into the baseline gaps, in ROM order. */
        asm/rom_header.o(.text);        /* 0x000000..0x0000BF */
        asm/baserom.o(.rom_0);          /* 0x0000C0..0x000B60 */
        src/rng.o(.text);               /* 0x000B60..0x000CD8 */
        asm/baserom.o(.rom_1);          /* 0x000CD8..0x0DC134 */
        src/rng.o(.rodata);             /* 0x0DC134..0x0DC144 */
        asm/baserom.o(.rom_2);          /* 0x0DC144..EOF       */
    } > rom

    /* RAM (NOLOAD) sections: not emitted to the ROM, but their symbols must
       resolve to the JP addresses that decompiled .text literals reference. */
    .bss 0x03000000 (NOLOAD) :
    {
        src/rng.o(.bss);                /* gRNSeeds @ 0x03000000, gLCGRNValue @ 0x03000008 */
        src/rng.o(COMMON);
    } > iwram

    /DISCARD/ :
    {
        *(.debug*)
        *(.comment)
        *(.note*)
        *(.ARM.attributes)
    }
}
